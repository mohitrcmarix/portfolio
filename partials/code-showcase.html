<?php
/**
 * Secure WordPress AJAX Endpoint for Dynamic Filtering
 * Developed by Mohit Rathod
 */
add_action('wp_ajax_filter_portfolio_projects', 'mohit_filter_portfolio_projects');
add_action('wp_ajax_nopriv_filter_portfolio_projects', 'mohit_filter_portfolio_projects');

function mohit_filter_portfolio_projects() {
    // Verify Security Nonce
    check_ajax_referer('mohit_portfolio_nonce', 'security');

    $category = isset($_POST['category']) ? sanitize_text_field($_POST['category']) : 'all';
    
    $args = [
        'post_type'      => 'portfolio_project',
        'posts_per_page' => 9,
        'post_status'    => 'publish',
    ];

    if ($category !== 'all') {
        $args['tax_query'] = [
            [
                'taxonomy' => 'project_category',
                'field'    => 'slug',
                'terms'    => $category,
            ]
        ];
    }

    $query = new WP_Query($args);
    $projects = [];

    if ($query->have_posts()) {
        while ($query->have_posts()) {
            $query->the_post();
            $projects[] = [
                'id'       => get_the_ID(),
                'title'    => get_the_title(),
                'excerpt'  => get_the_excerpt(),
                'live_url' => get_field('live_website_url'),
                'stack'    => get_field('tech_stack_tags'),
                'thumb'    => get_the_post_thumbnail_url(get_the_ID(), 'medium_large'),
            ];
        }
        wp_reset_postdata();
        wp_send_json_success(['projects' => $projects]);
    } else {
        wp_send_json_error(['message' => 'No projects found in this category.']);
    }
}

/**
 * WooCommerce Custom Dynamic Checkout Logic
 * Developed by Mohit Rathod
 */
add_action('woocommerce_cart_calculate_fees', 'mohit_custom_express_handling_fee');
function mohit_custom_express_handling_fee($cart) {
    if (is_admin() && !defined('DOING_AJAX')) {
        return;
    }

    // Check custom customer session flag
    $is_express = WC()->session->get('mohit_express_delivery');
    if ($is_express) {
        $fee_amount = 150.00; // INR express packaging
        $cart->add_fee(__('Express Priority Handling', 'mohit-theme'), $fee_amount, true, 'standard');
    }
}

// Razorpay Custom Payment Verification Hook
add_action('woocommerce_payment_complete', 'mohit_verify_razorpay_order_status');
function mohit_verify_razorpay_order_status($order_id) {
    $order = wc_get_order($order_id);
    if (!$order) return;

    if ($order->get_payment_method() === 'razorpay') {
        $razorpay_payment_id = $order->get_meta('_razorpay_payment_id');
        if (!empty($razorpay_payment_id)) {
            $order->add_order_note(
                sprintf(__('Payment verified via Razorpay ID: %s. Order moved to processing.', 'mohit-theme'), $razorpay_payment_id)
            );
        }
    }
}

/**
 * Core PHP OOP Database CRUD Class
 * Developed by Mohit Rathod
 */
class DatabaseManager {
    private PDO $pdo;

    public function __construct(string $host, string $db, string $user, string $pass) {
        $dsn = "mysql:host={$host};dbname={$db};charset=utf8mb4";
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        $this->pdo = new PDO($dsn, $user, $pass, $options);
    }

    public function insert(string $table, array $data): int {
        $fields = implode(', ', array_keys($data));
        $placeholders = ':' . implode(', :', array_keys($data));
        $sql = "INSERT INTO {$table} ({$fields}) VALUES ({$placeholders})";
        
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($data);
        return (int)$this->pdo->lastInsertId();
    }

    public function fetchAllWhere(string $table, string $column, $value): array {
        $sql = "SELECT * FROM {$table} WHERE {$column} = :val ORDER BY id DESC";
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute(['val' => $value]);
        return $stmt->fetchAll();
    }
}
